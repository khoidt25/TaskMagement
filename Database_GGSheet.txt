BEGIN;

-- ============================================================
-- 1. APP ROLES - quyền hệ thống
-- ============================================================

CREATE TABLE IF NOT EXISTS app_roles (
    role_id       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    role_code     VARCHAR(30) NOT NULL UNIQUE,
    role_name     VARCHAR(100) NOT NULL,
    description   TEXT,
    created_at    TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at    TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 2. USERS
-- ============================================================

CREATE TABLE IF NOT EXISTS users (
    user_id        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    username       VARCHAR(100) NOT NULL UNIQUE,
    password_hash  VARCHAR(255) NOT NULL,
    full_name      VARCHAR(200) NOT NULL,
    email          VARCHAR(255) UNIQUE,
    role_id        BIGINT NOT NULL REFERENCES app_roles(role_id),
    user_status    VARCHAR(20) NOT NULL DEFAULT 'ACTIVE'
                   CHECK (user_status IN ('ACTIVE', 'INACTIVE', 'LOCKED')),
    created_at     TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at     TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 3. PROJECTS
-- ============================================================

CREATE TABLE IF NOT EXISTS projects (
    project_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    project_code    VARCHAR(50) NOT NULL UNIQUE,
    project_name    VARCHAR(200) NOT NULL,
    description     TEXT,
    project_status  VARCHAR(20) NOT NULL DEFAULT 'ACTIVE'
                    CHECK (project_status IN ('ACTIVE', 'INACTIVE', 'ARCHIVED')),
    created_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 4. CUSTOMERS
-- ============================================================

CREATE TABLE IF NOT EXISTS customers (
    customer_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    customer_code    VARCHAR(50) NOT NULL UNIQUE,
    customer_name    VARCHAR(200) NOT NULL,
    description      TEXT,
    customer_status  VARCHAR(20) NOT NULL DEFAULT 'ACTIVE'
                     CHECK (customer_status IN ('ACTIVE', 'INACTIVE')),
    created_at       TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at       TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 5. TASK ROLES - Role trong từng Task
--    Khác với app_roles.
--
--    app_roles:
--       ADMIN / MANAGER / MEMBER
--
--    task_roles:
--       BRSE / DEV / TESTER / REVIEWER / PM ...
-- ============================================================

CREATE TABLE IF NOT EXISTS task_roles (
    task_role_id  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    role_code     VARCHAR(50) NOT NULL UNIQUE,
    role_name     VARCHAR(100) NOT NULL,
    description   TEXT,
    is_active     BOOLEAN NOT NULL DEFAULT TRUE,
    created_at    TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at    TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 6. PRIORITIES
--
-- Sheet hiện tại đang có Priority dạng số như 3.
-- Dùng master để vừa giữ giá trị số vừa có tên hiển thị.
-- ============================================================

CREATE TABLE IF NOT EXISTS priorities (
    priority_id    INTEGER PRIMARY KEY,
    priority_code  VARCHAR(30) NOT NULL UNIQUE,
    priority_name  VARCHAR(100) NOT NULL,
    display_order  INTEGER NOT NULL DEFAULT 0,
    is_active      BOOLEAN NOT NULL DEFAULT TRUE
);

select *from priorities

-- ============================================================
-- 7. ISSUE TYPES
--    Dùng cho loại Issue từ Redmine / hoặc App.
-- ============================================================

CREATE TABLE IF NOT EXISTS issue_types (
    issue_type_id  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    type_code      VARCHAR(50) NOT NULL UNIQUE,
    type_name      VARCHAR(100) NOT NULL,
    is_active      BOOLEAN NOT NULL DEFAULT TRUE,
    created_at     TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at     TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 8. REDMINE ISSUES
--
-- Lưu dữ liệu gốc đồng bộ từ Redmine.
-- Không trộn toàn bộ field Redmine vào tasks.
-- ============================================================

CREATE TABLE IF NOT EXISTS redmine_issues (
    redmine_issue_id      BIGINT PRIMARY KEY,

    issue_project_id      BIGINT,
    issue_project_code    VARCHAR(100),
    issue_project_name    VARCHAR(255),

    issue_type            VARCHAR(100),
    issue_title           VARCHAR(1000),

    issue_priority        VARCHAR(100),
    issue_status          VARCHAR(100),

    redmine_url           TEXT,

    redmine_created_at    TIMESTAMPTZ,
    redmine_updated_at    TIMESTAMPTZ,

    synced_at             TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 9. TASKS - bảng trung tâm
-- ============================================================

CREATE TABLE IF NOT EXISTS tasks (
    task_id             BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    -- ID hiển thị của App
    task_code           VARCHAR(50) UNIQUE,

    -- Nội dung Task
    task_name           VARCHAR(1000) NOT NULL,

    -- Master
    project_id          BIGINT REFERENCES projects(project_id) ON DELETE SET NULL,
    customer_id         BIGINT REFERENCES customers(customer_id) ON DELETE SET NULL,

    -- Người tạo
    created_by          BIGINT NOT NULL REFERENCES users(user_id),

    -- Redmine integration
    redmine_issue_id    BIGINT UNIQUE
                        REFERENCES redmine_issues(redmine_issue_id)
                        ON DELETE SET NULL,

    -- Date
    start_date          DATE,
    deadline            DATE,

    -- RiliDate / Release Date
    release_date        DATE,

    -- App Priority
    priority_id         INTEGER NOT NULL DEFAULT 2
                        REFERENCES priorities(priority_id),

    -- App Status
    status              VARCHAR(20) NOT NULL DEFAULT 'NOT_STARTED'
                        CHECK (
                            status IN (
                                'NOT_STARTED',
                                'IN_PROGRESS',
                                'PENDING',
                                'COMPLETED',
                                'CANCELLED'
                            )
                        ),

    -- AUTO = tính từ member
    -- MANUAL = Manager tự điều chỉnh
    status_source       VARCHAR(10) NOT NULL DEFAULT 'AUTO'
                        CHECK (status_source IN ('AUTO', 'MANUAL')),

    description         TEXT,
    note                TEXT,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT chk_tasks_dates
        CHECK (
            start_date IS NULL
            OR deadline IS NULL
            OR start_date <= deadline
        )
);

-- ============================================================
-- 10. TASK MEMBERS
--
-- Đây là bảng quan trọng nhất.
--
-- Một Task:
--   A = BrSE
--   B = Dev
--   C = Tester
--
-- Mỗi người có:
--   Role
--   Assigned Work
--   Status
--   Progress
--   Deadline
--   Note
-- ============================================================

CREATE TABLE IF NOT EXISTS task_members (
    task_member_id  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    task_id         BIGINT NOT NULL
                    REFERENCES tasks(task_id) ON DELETE CASCADE,

    user_id         BIGINT NOT NULL
                    REFERENCES users(user_id),

    task_role_id    BIGINT
                    REFERENCES task_roles(task_role_id)
                    ON DELETE SET NULL,

    assigned_work   TEXT NOT NULL,

    status          VARCHAR(20) NOT NULL DEFAULT 'NOT_STARTED'
                    CHECK (
                        status IN (
                            'NOT_STARTED',
                            'IN_PROGRESS',
                            'PENDING',
                            'COMPLETED',
                            'CANCELLED'
                        )
                    ),

    progress        SMALLINT NOT NULL DEFAULT 0
                    CHECK (progress BETWEEN 0 AND 100),

    deadline        DATE,

    note            TEXT,

    assigned_by     BIGINT
                    REFERENCES users(user_id)
                    ON DELETE SET NULL,

    assigned_at     TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    created_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uq_task_members_task_user
        UNIQUE (task_id, user_id)
);

-- ============================================================
-- 11. TASK REVIEWERS
--
-- Sheet hiện tại có:
--   Reviewer
--   ReviewStatus
--
-- Một Task có thể có nhiều Reviewer.
-- ============================================================

CREATE TABLE IF NOT EXISTS task_reviewers (
    task_reviewer_id  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    task_id           BIGINT NOT NULL
                      REFERENCES tasks(task_id) ON DELETE CASCADE,

    reviewer_id       BIGINT NOT NULL
                      REFERENCES users(user_id),

    review_status     VARCHAR(30) NOT NULL DEFAULT 'NOT_STARTED'
                      CHECK (
                          review_status IN (
                              'NOT_STARTED',
                              'IN_REVIEW',
                              'APPROVED',
                              'REJECTED'
                          )
                      ),

    note              TEXT,

    created_at        TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at        TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uq_task_reviewer
        UNIQUE (task_id, reviewer_id)
);

-- ============================================================
-- 12. TASK LINKS
--
-- Các link ngoài:
--   Redmine
--   Teams
--   Wiki
--   Other
--
-- Redmine chính thức có thể dùng redmine_issue_id ở tasks,
-- còn bảng này dùng để lưu URL chi tiết.
-- ============================================================

CREATE TABLE IF NOT EXISTS task_links (
    task_link_id  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    task_id       BIGINT NOT NULL
                  REFERENCES tasks(task_id) ON DELETE CASCADE,

    link_type     VARCHAR(30) NOT NULL
                  CHECK (
                      link_type IN (
                          'REDMINE',
                          'TEAMS',
                          'WIKI',
                          'OTHER'
                      )
                  ),

    link_name     VARCHAR(255),
    link_url      TEXT NOT NULL,

    created_by    BIGINT
                  REFERENCES users(user_id)
                  ON DELETE SET NULL,

    created_at    TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================
-- 13. TASK STATUS HISTORY
-- ============================================================

CREATE TABLE IF NOT EXISTS task_status_history (
    history_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    task_id      BIGINT NOT NULL
                 REFERENCES tasks(task_id) ON DELETE CASCADE,

    old_status   VARCHAR(20),

    new_status   VARCHAR(20) NOT NULL
                 CHECK (
                     new_status IN (
                         'NOT_STARTED',
                         'IN_PROGRESS',
                         'PENDING',
                         'COMPLETED',
                         'CANCELLED'
                     )
                 ),

    changed_by   BIGINT
                 REFERENCES users(user_id)
                 ON DELETE SET NULL,

    changed_at   TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    note         TEXT
);

-- ============================================================
-- 14. TASK MEMBER STATUS / PROGRESS HISTORY
-- ============================================================

CREATE TABLE IF NOT EXISTS task_member_status_history (
    history_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    task_member_id  BIGINT NOT NULL
                    REFERENCES task_members(task_member_id)
                    ON DELETE CASCADE,

    old_status      VARCHAR(20),

    new_status      VARCHAR(20) NOT NULL
                    CHECK (
                        new_status IN (
                            'NOT_STARTED',
                            'IN_PROGRESS',
                            'PENDING',
                            'COMPLETED',
                            'CANCELLED'
                        )
                    ),

    old_progress    SMALLINT
                    CHECK (old_progress BETWEEN 0 AND 100),

    new_progress    SMALLINT
                    CHECK (new_progress BETWEEN 0 AND 100),

    changed_by      BIGINT
                    REFERENCES users(user_id)
                    ON DELETE SET NULL,

    changed_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    note            TEXT
);

-- ============================================================
-- 15. DAILY WORK LOG
--
-- Dùng cho:
--   Công số dự kiến trong ngày
--   Công số thực tế
--
-- Ví dụ:
--   User A / Task 123 / 2026-10-08 / Planned 4h / Actual 3h
-- ============================================================

CREATE TABLE IF NOT EXISTS task_work_logs (
    work_log_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    task_member_id  BIGINT NOT NULL
                    REFERENCES task_members(task_member_id)
                    ON DELETE CASCADE,

    work_date       DATE NOT NULL,

    planned_hours   NUMERIC(5,2)
                    CHECK (
                        planned_hours IS NULL
                        OR planned_hours BETWEEN 0 AND 24
                    ),

    actual_hours    NUMERIC(5,2)
                    CHECK (
                        actual_hours IS NULL
                        OR actual_hours BETWEEN 0 AND 24
                    ),

    note            TEXT,

    created_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uq_task_work_logs_member_date
        UNIQUE (task_member_id, work_date)
);

-- ============================================================
-- 16. NOTIFICATIONS
-- ============================================================

CREATE TABLE IF NOT EXISTS notifications (
    notification_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    user_id           BIGINT NOT NULL
                      REFERENCES users(user_id)
                      ON DELETE CASCADE,

    task_id           BIGINT
                      REFERENCES tasks(task_id)
                      ON DELETE CASCADE,

    notification_type VARCHAR(40) NOT NULL
                      CHECK (
                          notification_type IN (
                              'TASK_ASSIGNED',
                              'DEADLINE_SOON',
                              'TASK_OVERDUE',
                              'TASK_UPDATED',
                              'MEMBER_WORK_UPDATED',
                              'STATUS_CHANGED',
                              'REVIEW_ASSIGNED',
                              'REVIEW_STATUS_CHANGED'
                          )
                      ),

    title             VARCHAR(255) NOT NULL,

    message           TEXT,

    is_read           BOOLEAN NOT NULL DEFAULT FALSE,

    created_at        TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    read_at           TIMESTAMPTZ
);
